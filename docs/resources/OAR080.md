Security requirements, global or per operation, must only use the security schemes defined in `securityDefinitions` (OpenAPI 2) or `components.securitySchemes` (OpenAPI 3).

Noncompliant Code Example (OpenAPI 2)
-------------------------------------

``` yaml
swagger: '2.0'
info:
  title: Customer Onboarding API
  version: 1.0.0
securityDefinitions:
  apiKeyAuth:
    type: apiKey
    in: header
    name: X-API-Key
security:
  - apiKeyAuth: []
  - partnerOAuth: [] # Noncompliant {{OAR080: Security scheme 'partnerOAuth' is not defined in securityDefinitions.}}
paths:
  /customers:
    get:
      security:
        - apiKeyAuth: []
      responses:
        '200':
          description: OK
    post:
      security:
        - basicAuth: [] # Noncompliant {{OAR080: Security scheme 'basicAuth' is not defined in securityDefinitions.}}
      responses:
        '201':
          description: Created
```

Compliant Solution (OpenAPI 2)
------------------------------

``` yaml
swagger: '2.0'
info:
  title: Customer Onboarding API
  version: 1.0.0
securityDefinitions:
  apiKeyAuth:
    type: apiKey
    in: header
    name: X-API-Key
  partnerOAuth:
    type: oauth2
    flow: application
    tokenUrl: https://auth.example.com/oauth/token
    scopes:
      customers:write: Manage customers
  basicAuth:
    type: basic
security:
  - apiKeyAuth: []
  - partnerOAuth: []
paths:
  /customers:
    get:
      security:
        - apiKeyAuth: []
      responses:
        '200':
          description: OK
    post:
      security:
        - basicAuth: []
      responses:
        '201':
          description: Created
```

Noncompliant Code Example (OpenAPI 3)
-------------------------------------

``` yaml
openapi: 3.0.3
info:
  title: Customer Onboarding API
  version: 1.0.0
security:
  - apiKeyAuth: []
paths:
  /customers:
    get:
      security:
        - apiKeyAuth: []
      responses:
        '200':
          description: OK
    post:
      security:
        - ApiKeyAuth: [] # Noncompliant {{OAR080: Security scheme 'ApiKeyAuth' is not defined in components.securitySchemes.}}
      responses:
        '201':
          description: Created
      callbacks:
        customerOnboarded:
          '{$request.body#/callbackUrl}':
            post:
              security:
                - signatureKey: [] # Noncompliant {{OAR080: Security scheme 'signatureKey' is not defined in components.securitySchemes.}}
              responses:
                '204':
                  description: Received
components:
  securitySchemes:
    apiKeyAuth:
      type: apiKey
      in: header
      name: X-API-Key
```

Compliant Solution (OpenAPI 3)
------------------------------

``` yaml
openapi: 3.0.3
info:
  title: Customer Onboarding API
  version: 1.0.0
security:
  - apiKeyAuth: []
paths:
  /customers:
    get:
      security:
        - apiKeyAuth: []
      responses:
        '200':
          description: OK
    post:
      responses:
        '201':
          description: Created
      callbacks:
        customerOnboarded:
          '{$request.body#/callbackUrl}':
            post:
              security:
                - signatureKey: []
              responses:
                '204':
                  description: Received
  /health:
    get:
      security: []
      responses:
        '200':
          description: OK
components:
  securitySchemes:
    apiKeyAuth:
      type: apiKey
      in: header
      name: X-API-Key
    signatureKey:
      type: apiKey
      in: header
      name: X-Signature
```
